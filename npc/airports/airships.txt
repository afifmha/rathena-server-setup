        mes "you can't participate in the game.";

        mes "Next, we will each roll the dice twice.";
        mes "The one with the higher sum of numbers";
        mes "wins. But deciding it at once isn't fun,";
        mes "so I will give both sides a chance to roll";
        mes "one more time.";

        mes "["+ .@npc_name$ +"]";
        mes "It's up to you whether to roll one more";
        mes "time or not, but if you roll again and";
        mes "the sum exceeds 12, the side that exceeds";
        mes "it loses.";

    }

    mes "You can bet up to 50,";
    mes "and if you want to cancel the game,";
